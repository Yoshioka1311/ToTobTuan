function มี 2 ประเภท: คืนค่า และ ไม่คืนค่า

ไม่คืนค่า (Not return value)
function showMessage() {
  alert('Hello everyone!');
}
คืนค่า (Return value)
function sum(a, b) {
  return a + b;
}
function showMessage() {
  console.log('Hello everyone!');   // สไลด์ใช้ alert() — ที่นี่ใช้ console.log เพื่อดูผลในกล่องด้านล่าง
}
function sum(a, b) {
  return a + b;
}

showMessage();
console.log(sum(3, 4));
console.log(showMessage());   // function ที่ไม่มี return จะได้ undefined

Parameters (Arguments) และค่า default

function ใน JS รองรับการส่งข้อมูลอะไรก็ได้เข้าไป ข้อมูลนั้นเรียกว่า Parameters หรือ Arguments และ parameter ตั้งค่า default ได้

ไม่มี default
function showMessagefromuser(from, text) {
  alert(from + ': ' + text);
}
มี default
function showMessagefromuser(
  from, text = "no message"
) {
  alert(from + ': ' + text);
}
function showMessagefromuser(from, text = "no message") {
  console.log(from + ': ' + text);
}

showMessagefromuser("Ann", "เจอกันพรุ่งนี้");
showMessagefromuser("Ann");            // ไม่ส่ง text → ใช้ค่า default

Arrow functions

arrow function คือไวยากรณ์ที่สั้นและกระชับสำหรับสร้าง function

function expression เดิม
let func = function (arg1, arg2, ...argN) {
  return expression;
};
arrow function
let func = (arg1, arg2, ...argN) => expression

ตัวอย่างจากสไลด์

// เขียนบรรทัดเดียว — ค่าของ expression คือค่าที่คืนออกมา (ไม่ต้องเขียน return)
let sum = (a, b) => a + b;

// ไม่มี parameter — ต้องมีวงเล็บว่าง ()
let sayHello = () => console.log("Hello!");   // สไลด์ใช้ alert("Hello!")

// หลายบรรทัด — ต้องมีปีกกา { } และเขียน return เอง
let subtract = (a, b) => {
  let result = a - b;
  return result;
};

console.log(sum(2, 3));
sayHello();
console.log(subtract(9, 4));
จุดที่มักพลาด

แบบบรรทัดเดียว (a, b) => a + b คืนค่าให้อัตโนมัติ แต่พอใส่ปีกกา { } แล้วต้องเขียน return เอง ไม่อย่างนั้นจะได้ undefined

First Class Functions

นิยามจากสไลด์

ภาษาหนึ่งจะเรียกว่ามี First-class Functions เมื่อ function ในภาษานั้นถูกปฏิบัติเหมือนตัวแปรอื่นๆ

คุณสมบัตินี้ถูกใช้ประโยชน์อย่างมากในกระบวนทัศน์การเขียนโปรแกรมที่ชื่อ functional programming

4 อย่างที่ first-class function ทำได้

1. assign function เข้าตัวแปร (anonymous function หรือ expression function)

// Create an anonymous function
var function2 = function (wide) {
  var area;
  area = Math.pow(wide, 2);
  return area;
};

console.log(function2(6));

2. ส่ง function เป็น argument ให้ function อื่น

function flexible(fOperation, operand1, operand2) {
  var result = fOperation(operand1, operand2);
  return result;
}

let sumFunction = function (num1, num2) {
  return num1 + num2;
};

let output = flexible(sumFunction, 3, 4);
console.log(output);

3. ให้ function คืนค่าเป็น function

const calculateAreaofSquare2 = function (wide) {
  var area;
  area = Math.pow(wide, 2);
  return () => {
    console.log(`calculateAreaofSquare2: ${area}`);
  };
};

let areaOfSquare = calculateAreaofSquare2(4);
areaOfSquare();

4. เก็บ function ไว้ในโครงสร้างข้อมูล

function calculateAreaofSquare(wide) {
  var area;
  area = Math.pow(wide, 2);
  return area;
}

let myFuncs = [calculateAreaofSquare];
console.log(`calculateAreaofSquare: ${myFuncs[0](5)}`);

High-order Function

นิยามจากสไลด์

High-order functions คือ function ที่รับ function อื่นเป็น argument หรือ คืนค่าผลลัพธ์เป็น function

ตัวอย่างในสไลด์ใช้ map และ filter ของ Array ซึ่งรับ function เข้าไปทำงานกับสมาชิก

function calculateAreaofSquare(wide) {
  var area;
  area = Math.pow(wide, 2);
  return area;
}

let squareWide1 = [2, 7, 9, 10];
console.log("Map: " + squareWide1.map(calculateAreaofSquare));

const evenNumbers = squareWide1.filter((num) => num % 2 === 0);
console.log(`evenNumbers: ${evenNumbers}`);

ไล่ทีละขั้น — map: 2² = 4, 7² = 49, 9² = 81, 10² = 100 จึงได้ 4,49,81,100 · filter เก็บเฉพาะตัวที่ num % 2 === 0 คือ 2 และ 10 จึงได้ 2,10

Callback Functions

จุดประสงค์หลัก

callback function มีไว้เป็นกลไกให้โปรแกรมตอบสนองต่อ event หรือการกระทำที่เกิดขึ้นแบบ asynchronous หรือคาดเดาเวลาไม่ได้

APIs (Application Programming Interfaces) คือ library ของ object และ method ที่ถูก implement ไว้แล้ว บางครั้ง function ของ API ต้องการ function เป็น parameter — เราต้องเขียน function นั้นเองโดยมี parameter list ตามที่เอกสารของ API กำหนด function แบบนี้เรียกว่า Callback function

function greet() {
  console.log('Howdy!');       // สไลด์ใช้ alert('Howdy!')
}

setTimeout(greet, 2000);       // greet คือ callback ที่ setTimeout จะเรียกให้เมื่อครบ 2 วินาที
console.log('ตั้งเวลาเรียบร้อย — บรรทัดนี้ทำงานก่อน');
สังเกตลำดับผลลัพธ์

บรรทัด "ตั้งเวลาเรียบร้อย" ขึ้นก่อน "Howdy!" เพราะ setTimeout ไม่หยุดรอ — ฝาก greet ไว้แล้วให้บรรทัดถัดไปทำงานต่อทันที นี่คือความหมายของ asynchronous ที่สไลด์พูดถึง

Scope

สไลด์เปรียบว่า function เหมือนแผนกในสำนักงาน

  • ตัวแปรใน JavaScript เป็น lexically scoped ยกเว้น var
  • ตัวแปรที่ประกาศนอก function ใดๆ → scope คือโค้ดทั้งหมดที่ผูกกับหน้าเว็บนั้น (Global variable)
  • ตัวแปรที่ประกาศใน function → scope คือทั้ง function นั้น (Local variable)
Note จากสไลด์

ตัวแปรควรมี scope ที่เล็กที่สุด

Block scope: ตัวอย่างและ error ที่ได้

myName ในสคริปต์นี้เป็น block-scoped กับคำสั่ง if — เข้าถึงได้เฉพาะภายใน block เท่านั้น

<script type="text/javascript">
  if (true) {
    const myName = "Wudhichart";
    console.log(myName);
  }
</script>

ถ้าลองเข้าถึง myName นอก block จะได้ ReferenceError เพราะมันอยู่นอก scope

if (true) {
  const myName = "Wudhichart";
  console.log(myName);
}
console.log(myName);   // Uncaught ReferenceError: myName is not defined

scope ของ let, const และ var

  • let และ const สร้างตัวแปรแบบ block-scoped
  • var สร้างตัวแปรที่ผูกกับ function ที่ใกล้ที่สุด หรือ global scope
function displayFullname() {
  if (true) {
    var myFullname = "Wudhichart Sawangphol";
  }
  console.log(myFullname);   // ไม่ error เพราะ var ผูกกับทั้ง function ไม่ใช่แค่ block
}
displayFullname();

โค้ดชุดเดียวกันนี้ ถ้าเปลี่ยน var เป็น let หรือ const จะได้ ReferenceError ทันที — แก้ในช่องโค้ดแล้วกดรันเทียบได้

Global Scope และ Module Scope

Global Scope — ข้าม <script> ได้
<script type="text/javascript">
  const myFullname = "Wudhichart Sawangphol";
</script>
<script type="text/javascript">
  console.log(myFullname);
</script>
Module Scope — แต่ละ module มี scope ของตัวเอง
<script type="module">
  const jidapa = "Jidapa Kraisangka";
</script>
<script type="text/javascript">
  console.log(jidapa);
  // Uncaught ReferenceError: jidapa is not defined
</script>

Lifetime (อายุของตัวแปร)

Global variables
ตั้งแต่ถูกประกาศ จนถึงหน้าเว็บถูกปิด
Local variables
ตั้งแต่ function ที่ประกาศมันถูกเรียก จนถึง function นั้นทำงานจบ
ชื่อชนกัน

ถ้า global variable กับ local variable ชื่อเดียวกัน ตัวที่ถูกใช้คือ local variable

Closures

นิยามจากสไลด์

Closure คือ function ที่รวมเข้ากับการอ้างอิงตัวแปรที่ถูกนิยามไว้ข้างนอกตัวมัน — closure รักษาการอ้างอิงตัวแปรเหล่านั้นไว้ ทำให้ function เข้าถึงตัวแปรนอก scope ของตัวเองได้

// Closures
function displayFullname(name) {
  return () => console.log(name);
}

const printFullname = displayFullname("Wudhichart Sawangphol");
printFullname();

แม้ displayFullname จะทำงานจบไปแล้ว แต่ arrow function ที่คืนออกมายังอ้างถึงตัวแปร name ได้อยู่ — นี่คือสิ่งที่ closure ทำ

OOP คืออะไร Class และ Object ประกาศ class ใน JS สร้าง object Inheritance และ Polymorphism เพิ่ม property ทีหลัง Constructor Functions

OOP คืออะไร

  • ใช้แนวคิดเดียวกับที่เราแทนสิ่งของในชีวิตจริง เช่น รถยนต์ คน บัญชีธนาคาร
  • เป็นการนิยามชนิดข้อมูลของโครงสร้างข้อมูล และชนิดของ function ที่ใช้กับโครงสร้างข้อมูลนั้นได้

Class และ Object

Class

แม่แบบ (template) หรือโมเดลของ object ประกอบด้วย properties และ functions

Object

คือ class ที่ถูกเติมค่าลงไปแล้ว — เมื่อ object ถูก instantiate จาก class มันจะมี properties และ methods ทั้งหมดที่นิยามไว้ใน class

ประกาศ class ใน JavaScript

class className {
  constructor(prop1, prop2, …) {
    /* Property definition and assignment */
  }
  /* Method definition */
  method1() { /* do something */ }
  method2() { /* do something */ }
}
constructor

method ชื่อ constructor จะถูกเรียกทุกครั้งที่ object ถูกสร้าง (initiate)

ตัวอย่างจากสไลด์: class Instructor

class Instructor {
  /* Constructor */
  constructor(_firstName, _lastName, _subjects) {
    /* Properties */
    this.firstName = _firstName;
    this.lastName  = _lastName;
    this.subjects  = _subjects;
  }
  /* Methods */
  sayHello() {
    return "Hey! I'm " + this.firstName + ". Nice to meet you!!!";
  }
  teach() {
    return "I teach " + this.subjects + ".";
  }
}

Note จากสไลด์: อย่าลืมเขียนไว้ใน <script type="application/javascript">…</script>

สร้าง object และเรียกใช้

  • class ต้องถูกนิยามก่อนที่ object จะถูกสร้าง
  • ใช้ new เรียก constructor ของ class เพื่อสร้าง object พร้อมค่าที่ระบุ
let obj = new className(prop1, prop2, …);

/* Get a value of a property */
console.log(obj.propName);

/* Assign a new value to a property */
obj.propName = newValue;

/* Call a method */
obj.methodName();

ตัวอย่างเต็มจากสไลด์ — กดรันดูผลได้

class Instructor {
  constructor(_firstName, _lastName, _subjects) {
    this.firstName = _firstName;
    this.lastName  = _lastName;
    this.subjects  = _subjects;
  }
  sayHello() { return "Hey! I'm " + this.firstName + ". Nice to meet you!!!"; }
  teach()    { return "I teach " + this.subjects + "."; }
}

/* Initiate an object of the Instructor class */
let jkSubjects = ["HTML", "OOP:JS"];
let kjidapa = new Instructor("Jidapa", "Kraisangka", jkSubjects);

/* Display firstName and lastName in the console */
console.log(kjidapa.firstName + " " + kjidapa.lastName);

/* Change firstName from "Jidapa" to "Aj. Pa" */
kjidapa.firstName = "Aj. Pa";
console.log(kjidapa.firstName);

/* Call sayHello() and teach() and display in the console */
console.log(kjidapa.sayHello());
console.log(kjidapa.teach());

Inheritance และ Polymorphism

  • object อาจมี properties และ methods บางอย่างต่างไปจากที่ได้รับมาจาก superclass
  • subclass สืบทอด properties และ methods ทั้งหมดของ superclass (inheritance) แต่มี properties และ methods เฉพาะของตัวเองได้ (polymorphism)
  • ใช้ extends เพื่อสร้าง subclass ที่สืบทอดจาก superclass
class subclassName extends superclassName {
  constructor(cprop1, cprop2, …, scprop1, scprop2, …) {
    /* Properties inherited from the superclass */
    super(cprop1, cprop2, …);
    /* Properties of the subclass */
    this.scprop1 = scprop1;
    …
  }
  /* Superclass's methods also inherited from the superclass */
  /* Only define methods that are unique for the subclass */
  subclassMethod1() { /* do something here */ }
}

ตัวอย่าง UnivInstructor ที่สืบทอดจาก Instructor

class Instructor {
  constructor(_firstName, _lastName, _subjects) {
    this.firstName = _firstName;
    this.lastName  = _lastName;
    this.subjects  = _subjects;
  }
  sayHello() { return "Hey! I'm " + this.firstName + ". Nice to meet you!!!"; }
  teach()    { return "I teach " + this.subjects + "."; }
}

/* Superclass: Instructor, Subclass: UnivInstructor */
class UnivInstructor extends Instructor {
  constructor(_firstName, _lastName, _subjects, _university, _campus) {
    /* Properties from the superclass */
    super(_firstName, _lastName, _subjects);
    /* Properties only in the subclass */
    this.university = _university;
    this.campus     = _campus;
  }
  /* Methods also inherited from the superclass */
  getUniversity() {
    return "My office is at " + this.university + ", " + this.campus + ".";
  }
}

let jkSubjects = ["HTML", "OOP:JS"];
let kjidapa2 = new UnivInstructor("Jidapa", "Kraisangka", jkSubjects,
                                  "Mahidol University", "Salaya");

console.log(kjidapa2.firstName + "@" + kjidapa2.university);
/* All methods from the superclass are inherited */
console.log(kjidapa2.sayHello());
console.log(kjidapa2.teach());
/* Call getUniversity() method */
console.log(kjidapa2.getUniversity());

Note จากสไลด์: class Instructor ต้องถูกนิยามก่อนนิยาม subclass UnivInstructor

เพิ่ม property และ method ให้ object ทีหลังได้

สไลด์ทดลองว่า ถ้าเรียก property หรือ function ที่ยังไม่ถูกนิยาม จะเกิดอะไรขึ้น

class Instructor {
  constructor(_firstName, _lastName, _subjects) {
    this.firstName = _firstName;
    this.lastName  = _lastName;
    this.subjects  = _subjects;
  }
  sayHello() { return "Hey! I'm " + this.firstName + ". Nice to meet you!!!"; }
  teach()    { return "I teach " + this.subjects + "."; }
}
class UnivInstructor extends Instructor {
  constructor(_firstName, _lastName, _subjects, _university, _campus) {
    super(_firstName, _lastName, _subjects);
    this.university = _university;
    this.campus     = _campus;
  }
  getUniversity() { return "My office is at " + this.university + ", " + this.campus + "."; }
}
let kjidapa2 = new UnivInstructor("Jidapa", "Kraisangka", ["HTML","OOP:JS"],
                                  "Mahidol University", "Salaya");

/* What if we try adding an undefined property */
console.log(kjidapa2.nickName);        // undefined
kjidapa2.nickName = "Pa";              // เพิ่ม property ทีหลัง
console.log(kjidapa2.nickName);        // Pa

/* เพิ่ม function ทีหลัง */
kjidapa2.sayMyName = function () {
  return "I am " + this.firstName + " or " + this.nickName;
};
console.log(kjidapa2.sayMyName());
ต่างกันตรงนี้

อ่าน property ที่ยังไม่มี → ได้ undefined (ไม่ error) แต่เรียก function ที่ยังไม่มี เช่น kjidapa2.sayMyName() ก่อนกำหนดค่า → เกิด error

Constructor Functions

ปกติเรามักอยากสร้าง object หลายตัวที่มีชุด properties เหมือนกัน ใน OOP เพราะจะประมวลผลแบบเดียวกันได้โดยไม่ผิดพลาด Constructor functions คือกลไกสร้าง object ที่มีชุด properties เหมือนกัน และสร้างได้ในแบบเดียวกับ function ปกติ

  • เรียกด้วย operator new
  • ใน constructor function ใช้ keyword this เพื่ออ้างถึงobject ที่กำลังถูกสร้าง
  • object ที่สร้างใหม่ทุกตัวได้รับชุด properties ที่สอดคล้องกัน
  • กำหนดค่าเริ่มต้นให้ properties ได้ผ่าน parameters ของ function
  • ใส่ function แบบ public (ใช้ this) และ private (ใช้ var, let) ใน constructor function ได้
function Person(firstName, lastName, age) {
  this.firstName = firstName;
  this.lastName = lastName;
  this.age = age;
  this.getInfo = function () {
    return this.firstName + " " + this.lastName + " Age is " + this.age;
  };
}

var person1 = new Person("Wudhichart", "Sawangphol", 30);
console.log(person1.getInfo());

01สไลด์แบ่ง function ใน JavaScript ออกเป็น 2 ประเภทตามเกณฑ์ใด?

สไลด์ระบุว่า there are two types of function: functions that do not return values และ functions that do return values

02function showMessagefromuser(from, text = "no message") เมื่อเรียกด้วย showMessagefromuser("Ann") จะแสดงผลว่าอะไร?

parameter ตั้งค่า default ได้ — เมื่อไม่ส่ง text เข้ามา ค่าที่ใช้คือ "no message"

03arrow function ที่ไม่มี parameter เลย ต้องเขียนอย่างไร?

แม้ไม่มี parameter ก็ต้องมีวงเล็บว่าง () ก่อนเครื่องหมาย =>

04arrow function ที่เขียนหลายบรรทัดด้วยปีกกา { } ต้องทำอะไรเพิ่มเมื่อเทียบกับแบบบรรทัดเดียว?

แบบ (a, b) => a + b คืนค่าให้อัตโนมัติ แต่พอใส่ { } แล้วต้องเขียน return เอง เช่น let subtract = (a, b) => { let result = a - b; return result; };

05ภาษาหนึ่งจะเรียกว่ามี First-class Functions เมื่อใด?

นิยามจากสไลด์: a programming language is said to have First-class Functions when functions in that language are treated like any other variable

06High-order function คือ function แบบใด?

นิยามจากสไลด์: high-order functions are functions that can take other functions as arguments or return functions as their results เช่น map และ filter

07เมื่อ squareWide1 = [2, 7, 9, 10] ผลของ squareWide1.filter((num) => num % 2 === 0) คืออะไร?

filter เก็บเฉพาะสมาชิกที่เงื่อนไขเป็นจริง — เลขที่หาร 2 ลงตัวในชุดนี้คือ 2 และ 10

08จุดประสงค์หลักของ callback function ตามสไลด์คืออะไร?

สไลด์ระบุว่า the primary purpose of a callback function is to provide a mechanism for a program to respond to events or actions that occur asynchronously or unpredictably

09โค้ด setTimeout(greet, 2000); console.log('ตั้งเวลาเรียบร้อย'); จะให้ผลลัพธ์ตามลำดับใด?

setTimeout ไม่หยุดรอ — มันฝาก greet ไว้เป็น callback แล้วให้บรรทัดถัดไปทำงานต่อทันที นี่คือความหมายของ asynchronous

10ข้อใดอธิบาย scope ของ let, const และ var ได้ถูกต้อง?

สไลด์ระบุว่า let and const create block-scoped variables ส่วน var creates a variable that is scoped to their nearest function or the global scope

11ถ้าประกาศ const myName ไว้ใน block ของ if แล้วเรียก console.log(myName) นอก block จะเกิดอะไรขึ้น?

myName เป็น block-scoped กับคำสั่ง if — เข้าถึงได้เฉพาะภายใน block เท่านั้น นอก block จะได้ Uncaught ReferenceError: myName is not defined

12ถ้าชื่อของ global variable กับ local variable ตรงกัน ตัวใดจะถูกใช้?

สไลด์ระบุว่า if the names of global variable and local variable are the same, the local variable is considered

13Closure ใน JavaScript คืออะไร?

นิยามจากสไลด์: a closure is a function combined with references to the variables defined outside of it — closures maintain the variable references

14method ชื่อ constructor ใน class ถูกเรียกเมื่อใด?

สไลด์ระบุว่า the constructor method is called each time the object is initiated

15ในการสร้าง subclass ที่สืบทอดจาก superclass ต้องใช้อะไร และ super(...) ทำหน้าที่อะไร?

สไลด์ใช้ class subclassName extends superclassName และเรียก super(cprop1, cprop2, …) ใน constructor สำหรับ properties inherited from the superclass

16ถ้าเรียกอ่าน property ของ object ที่ยังไม่เคยถูกนิยามไว้ เช่น console.log(kjidapa2.nickName) จะได้อะไร?

อ่าน property ที่ยังไม่มีได้ undefined (และเพิ่มให้ทีหลังได้แบบ on the fly) ต่างจากการเรียก function ที่ยังไม่มี ซึ่งจะเกิด error

17DOM เป็นมาตรฐานขององค์กรใด และคืออะไร?

สไลด์ระบุว่า the DOM is a W3C standard และ DOM is a programming interface for web documents ที่ให้เข้าถึงและอัปเดตเนื้อหา โครงสร้าง และ style ได้แบบ dynamic

18ในโครงสร้าง HTML DOM ทุก HTML tag ถือเป็น node ชนิดใด?

ทั้งเอกสารคือ document node · ทุก HTML tag คือ element node · ข้อความใน element คือ text node · ทุก attribute คือ attribute node · comment คือ comment node

19document.querySelector() คืนค่าอะไร?

querySelector คืน element แรกที่ตรง ส่วน querySelectorAll คืน static NodeList ของทุก element ที่ตรง

20ถ้าต้องการสร้าง <p> ใหม่พร้อมข้อความแล้วเพิ่มเข้าไปเป็นลูกตัวสุดท้ายของ <div> ควรใช้ชุด method ใด?

สไลด์ใช้ createElement() คู่กับ createTextNode() เพื่อสร้าง node แล้วใช้ appendChild() เพื่อเพิ่ม node ใหม่เป็นลูกตัวสุดท้ายของ parent
